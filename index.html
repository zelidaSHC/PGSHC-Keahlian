/************************************************************ * FORMAT & VALIDATE NOMBOR TELEFON MALAYSIA ************************************************************/ function formatMalaysiaPhone(phone) { let value = String(phone || '').trim(); // Buang space, dash, bracket value = value.replace(/[\s\-().]/g, ''); // +60123456789 -> 60123456789 if (value.startsWith('+60')) { value = value.substring(1); } // 0123456789 -> 60123456789 else if (value.startsWith('0')) { value = '60' + value.substring(1); } // 123456789 -> 60123456789 else if (value.startsWith('1')) { value = '60' + value; } // Buang semua character selain nombor value = value.replace(/\D/g, ''); /* * Format mobile Malaysia: * 601XXXXXXXX * atau * 601XXXXXXXXX */ if (!/^601\d{8,9}$/.test(value)) { return { valid: false, phone: value, message: 'No. telefon tidak sah. Sila semak nombor telefon ahli dalam Google Sheets.' }; } return { valid: true, phone: value, message: '' }; } /************************************************************ * HANTAR STATUS WHATSAPP ************************************************************/ async function sendStatusWhatsApp(row) { const button = document.getElementById( 'wa-' + row ); if (button) { button.disabled = true; button.innerText = '⏳ Semak nombor...'; } try { const key = sessionStorage.getItem( 'standhc_admin_key' ); if (!key) { throw new Error( 'Sesi Admin telah tamat. Sila login semula.' ); } /****************************************************** * AMBIL DATA TERKINI DARIPADA GOOGLE SHEETS ******************************************************/ const url = GOOGLE_SCRIPT_URL + '?action=members&key=' + encodeURIComponent(key); const result = await jsonpRequest(url); if (!result.ok) { throw new Error( result.error || 'Gagal mendapatkan data ahli.' ); } /****************************************************** * CARI AHLI ******************************************************/ const member = (result.members || []).find( function(item) { return Number(item.row) === Number(row); } ); if (!member) { throw new Error( 'Rekod ahli tidak dijumpai.' ); } /****************************************************** * SEMAK STATUS ******************************************************/ if ( member.status !== 'Diluluskan' && member.status !== 'Ditolak' ) { throw new Error( 'WhatsApp hanya boleh dihantar selepas status Diluluskan atau Ditolak.' ); } /****************************************************** * SEMAK NOMBOR TELEFON ******************************************************/ if (!member.telefon) { throw new Error( 'No. telefon ahli tiada dalam rekod.' ); } const phoneResult = formatMalaysiaPhone( member.telefon ); if (!phoneResult.valid) { throw new Error( 'No. telefon "' + member.telefon + '" tidak sah. Contoh nombor yang betul: 0123456789' ); } const phone = phoneResult.phone; /****************************************************** * BINA MESEJ ******************************************************/ let message = ''; /****************************************************** * DILULUSKAN ******************************************************/ if ( member.status === 'Diluluskan' ) { message = `🤾‍♂️ STAND HANDBALL CLUB (STAND HC) Assalamualaikum dan Selamat Sejahtera. 🎉 TAHNIAH! Terima kasih. 🤾‍♂️ STAND HC`; } /****************************************************** * BUKA WHATSAPP ******************************************************/ const whatsappUrl = 'https://wa.me/' + phone + '?text=' + encodeURIComponent( message ); window.open( whatsappUrl, '_blank' ); /****************************************************** * BUTTON ******************************************************/ if (button) { button.disabled = true; button.innerText = '✅ WhatsApp Dibuka'; } } catch (error) { if (button) { button.disabled = false; button.innerText = '📱 Hantar Status WhatsApp'; } alert( '❌ ' + error.message ); } }
